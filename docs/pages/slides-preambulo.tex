% Blocos de código — extensão da marca

\lstset{
    inputencoding=utf8,
    extendedchars=true,
    literate=%
      {á}{{\'a}}1 {à}{{\`a}}1 {ã}{{\~a}}1 {â}{{\^a}}1
      {é}{{\'e}}1 {ê}{{\^e}}1 {í}{{\'i}}1
      {ó}{{\'o}}1 {õ}{{\~o}}1 {ô}{{\^o}}1
      {ú}{{\'u}}1 {ç}{{\c c}}1
      {Á}{{\'A}}1 {É}{{\'E}}1 {Í}{{\'I}}1 {Ó}{{\'O}}1 {Ú}{{\'U}}1 {Ç}{{\c C}}1
}
\lstdefinestyle{sqlslide}{
    language         = SQL,
    morekeywords     = {AUTO_INCREMENT,ENGINE,InnoDB,IF,EXISTS,DATABASE,DATABASES,
                        REFERENCES,CONSTRAINT,CASCADE,BOOLEAN,TINYINT,VARCHAR,
                        TIMESTAMP,CHARACTER,COLLATE,LIMIT,SHOW,DESCRIBE,USE,
                        DECIMAL,TRUE,FALSE},
    basicstyle       = \ttfamily\footnotesize\color{white},
    keywordstyle     = \color{lightpurple}\bfseries,
    commentstyle     = \color{codecomment}\itshape,
    stringstyle      = \color{yellow},
    backgroundcolor  = \color{codebg},
    frame            = leftline,
    framesep         = 6pt,
    rulecolor        = \color{lightpurple},
    framerule        = 2pt,
    xleftmargin      = 8pt,
    framexleftmargin = 8pt,
    breaklines       = true,
    showstringspaces = false,
    upquote          = true,
    tabsize          = 2,
    aboveskip        = 0.6em,
    belowskip        = 0.4em,
}
